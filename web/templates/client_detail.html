      Auto-start Duckway on Linux with systemd
      

        
Run this on the client after the setup command above. It starts the local proxy at login; enable linger if it should run after boot before you log in.

        
# Finish client state first
duckway sync

mkdir -p ~/.config/systemd/user

cat > ~/.config/systemd/user/duckway-proxy.service <<'EOF'
[Unit]
Description=Duckway local API proxy
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
ExecStart=/usr/local/bin/duckway proxy
Restart=on-failure
RestartSec=5
Environment=HOME=%h

[Install]
WantedBy=default.target
EOF

cat > ~/.config/systemd/user/duckway-cc-watch.service <<'EOF'
[Unit]
Description=Duckway Control Channel watcher
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
ExecStart=/usr/local/bin/duckway cc watch
Restart=on-failure
RestartSec=5
Environment=HOME=%h

[Install]
WantedBy=default.target
EOF

systemctl --user daemon-reload
systemctl --user enable --now duckway-proxy

# Optional: only needed for Discord/Control Channel agents.
systemctl --user enable --now duckway-cc-watch

# Optional: keep user services running after boot before login.
sudo loginctl enable-linger "$USER"

        
If Duckway was installed somewhere other than /usr/local/bin/duckway, replace that path in both service files. Logs: journalctl --user -u duckway-proxy -f and journalctl --user -u duckway-cc-watch -f.
